### Experiment 22 – Palindrome Program in Java Using Docker

File: `Palindrome.java`

```java
public class Palindrome {
    public static void main(String[] args) {

        int n = 121, original = n, rev = 0;

        while (n > 0) {
            rev = rev * 10 + n % 10;
            n = n / 10;
        }

        if (original == rev)
            System.out.println("Palindrome");
        else
            System.out.println("Not Palindrome");
    }
}
```

File: `Dockerfile`

```dockerfile
FROM eclipse-temurin:21-jdk

CMD ["java", "Palindrome"]
```

```yaml
  name: event-app
spec:
  replicas: 2
  selector:
    matchLabels:
      app: event
  template:
    metadata:
      labels:
        app: event
    spec:
      containers:
      - name: web
        image: USER/event-app:v1
        ports:
        - containerPort: 80

---
apiVersion: v1
kind: Service
metadata:
  name: event-service
spec:
  selector:
    app: event
  ports:
```

```java
import org.openqa.selenium.*;
import org.openqa.selenium.edge.EdgeDriver;

public class FormTest {

    public static void main(String[] args) {

        WebDriver driver = new EdgeDriver();

        driver.get("http://localhost:8082/");

        WebElement name = driver.findElement(By.id("name"));

        WebElement phone = driver.findElement(By.id("phNo"));

        // TC1: Valid name
        name.sendKeys("Salman");

        if (name.getDomProperty("validationMessage").isEmpty())
            System.out.println("TC1: Passed");
        else
            System.out.println("TC1: Failed");

        // TC2: Invalid phone number
        phone.sendKeys("123");

        if (!phone.getDomProperty("validationMessage").isEmpty())
            System.out.println("TC2: Passed");
        else
            System.out.println("TC2: Failed");

        // TC3: Valid phone number
        phone.clear();

        phone.sendKeys("1234567890");

        if (phone.getDomProperty("validationMessage").isEmpty())
            System.out.println("TC3: Passed");
        else
            System.out.println("TC3: Failed");

        driver.quit();
    }
}
```

```text
TC2: Passed
TC3: Passed
```
